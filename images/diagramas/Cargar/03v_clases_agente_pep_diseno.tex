\documentclass[tikz,border=4pt,12pt]{standalone}
\input{estilos_uml}

  rotulo/.style={font=\ttfamily\footnotesize, fill=white, inner sep=1pt},
}
\begin{document}
\begin{tikzpicture}
% columnas a 5.6 cm, filas a 4.6 cm
% ===================== FILA 1: configuración, orquestación, cliente HTTP, buffer local
\umlclase[clase=4.4cm, fill=cPep, anchor=north]{cfg}{0,0}{dataclass}{AgentConfig}{%

\umlclase[clase=4.4cm, fill=cPep, anchor=north]{main}{5.8,0}{module}{main}{\vacio}{%

\umlclase[clase=4.4cm, fill=cPep, anchor=north]{buf}{17.4,0}{module}{event\_buffer}{\vacio}{%
 + append\_event()\\
 + flush\_pending()}

% ===================== FILA 2: creados por main
\node[clase, rectangle split parts=2, fill=cPep, text width=4.4cm, anchor=north,
  rectangle split part align={center,left}] (col) at (0,-4.8)
  {\makebox[\linewidth][c]{\textit{«colectores»}}\nodepart{two}\begin{tabular}{@{}l@{}}FileSystemCollector\\ProcessCollector\\AppAccessCollector\\NetworkCollector\end{tabular}};

% ===================== FILA 3: señal y política
\umlclase[clase=4.4cm, fill=cPep, anchor=north]{signal}{5.8,-10.4}{dataclass}{Signal}{%
 + event\_type\\

\umlclase[clase=4.4cm, fill=cPep, anchor=north]{pol}{17.4,-10.4}{dataclass}{EnforcementPolicy}{%

% ===================== RELACIONES
\draw[usa] (main.west) -- node[rotulo,above]{«usa»} (cfg.east |- main.west);
\draw[usa] (main.east) -- node[rotulo,above]{«crea»} (client.west |- main.east);

\node[rotulo, anchor=south west] at ([xshift=0.2cm]bus) {«crea»};

\draw[compone] (enf.south) -- node[rotulo,right]{1} (pol.north);
% señales
